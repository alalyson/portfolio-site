FORM save_deposit .
  DATA : ls_save  TYPE zte2fi0012,
         lt_save  LIKE TABLE OF ls_save,
         lv_tabix TYPE sy-index,
         ls_item_tmp TYPE zse2fi0001.
  CLEAR : lv_tabix, gt_docs.
  LOOP AT gt_deposit INTO gs_deposit.
    lv_tabix = sy-tabix.
    MOVE-CORRESPONDING gs_deposit TO ls_save.
    ls_save-seq_no = lv_tabix.
    ls_save-wrbtr  = ls_save-d_amt.
    ls_save-status = 'N'.
    IF ls_save-erdat IS INITIAL.
      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.
    ELSE.
      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.
    ENDIF.
    CLEAR gs_current_doc.
    gs_current_doc-header-bukrs = '1000'.
    gs_current_doc-header-gjahr = sy-datum(4).
    gs_current_doc-header-blart = 'DZ'. " 입금
    gs_current_doc-header-bldat = gs_deposit-ban_dat.
    gs_current_doc-header-budat = gs_deposit-ban_dat.
    gs_current_doc-header-waers = gs_deposit-waers.
    gs_current_doc-header-xblnr = ls_save-seq_no. " 참조번호에 일련번호 매핑
    ls_item_tmp = VALUE #( bschl = '40'
                           shkzg = 'S'
                           hkont = '0010000002'
                           wrbtr = gs_deposit-d_amt
                           sgtxt = |[고객 입금] { gs_deposit-contact }| ).
    APPEND ls_item_tmp TO gs_current_doc-items.
    ls_item_tmp = VALUE #( bschl = '50'
                           shkzg = 'H'
                           hkont = '0020000004' " 임시 지정한 가수금 계정
                           wrbtr = gs_deposit-d_amt
                           sgtxt = |[입금 가계정] { gs_deposit-contact }| ).
    APPEND ls_item_tmp TO gs_current_doc-items.
    APPEND gs_current_doc TO gt_docs.
    APPEND ls_save TO lt_save.
    CLEAR  ls_save.
  ENDLOOP.
  IF gt_docs IS NOT INITIAL.
    CALL METHOD zcl_e2_fi_posting=>create_fi_multi_document
      EXPORTING
        iv_wrttp      = '04' " 실제 데이터
      IMPORTING
        ev_last_belnr = gv_last_belnr
        et_fi_head    = gt_fi_head
        et_fi_item    = gt_fi_item
      CHANGING
        ct_docs       = gt_docs.
    LOOP AT lt_save ASSIGNING FIELD-SYMBOL(<fs_save>).
      READ TABLE gt_fi_head INTO DATA(ls_head_map) WITH KEY xblnr = <fs_save>-seq_no.
      IF sy-subrc = 0.
        <fs_save>-belnr = ls_head_map-belnr.
      ENDIF.
    ENDLOOP.
  ENDIF.
  IF lt_save IS NOT INITIAL.
    MODIFY zte2fi0012 FROM TABLE lt_save.
    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.
      MESSAGE s002.
    ENDIF.
  ENDIF.
ENDFORM.
